POST /api/v1/history
Content-Type: application/json

{
    "carId": 6,
    "bookingId": 1,
    "startMileage": 22000,
    "startDate": "2026-08-01 10:00:00",
    "endDate": "2026-08-04 10:00:00",
    "conditionBefore": "Excellent",
    "notes": "Автомобиль в отличном состоянии"
}
